Service Comparison

Service Comparison

How do GRIT, CIT, and AICommons compare as AI platforms? This page examines scores by category, the speed-vs-accuracy trade-off, and — critically — how the same model behaves when hosted on different infrastructure.

Infrastructure context: GRIT and CIT run all models locally via Ollama on UCSB-managed hardware. AICommons routes requests through Amazon Bedrock (AWS-hosted). This matters: the same model binary running on different hardware produces identical accuracy scores but dramatically different throughput and latency.

Average Score by Category

Mean score (1–5) per service per benchmark category, averaged across all models on that service. Scores use the canonical judge (Claude for literature synthesis).

Avg Score per Category, by Service

Bar height = mean score across all models on that service. Max = 5.0
{
  // Aggregate: service × category → mean avg_score
  const agg = {};
  for (const row of canonical_summary) {
    const k = row.service + "|||" + row.category;
    if (!agg[k]) agg[k] = { scores: [], service: row.service, category: row.category };
    agg[k].scores.push(row.avg_score);
  }

  const data = Object.values(agg).map(d => ({
    service:  d.service,
    category: d.category,
    cat_label: cat_labels_map[d.category] || d.category,
    mean_score: d.scores.reduce((a,b)=>a+b,0) / d.scores.length
  }));

  // Only categories with data
  const cats_present = [...new Set(data.map(d=>d.category))];
  const cat_order = cats_all.filter(c => cats_present.includes(c));

  const width  = 820;
  const height = 340;
  const marginL = 48, marginB = 90, marginT = 20, marginR = 20;
  const plotW = width - marginL - marginR;
  const plotH = height - marginT - marginB;

  // Outer scale: categories
  const x0 = d3.scaleBand()
    .domain(cat_order)
    .range([0, plotW])
    .paddingInner(0.25);

  // Inner scale: services
  const x1 = d3.scaleBand()
    .domain(svcs)
    .range([0, x0.bandwidth()])
    .paddingInner(0.08);

  const y = d3.scaleLinear().domain([0, 5]).range([plotH, 0]);

  const svg = d3.create("svg")
    .attr("width", width).attr("height", height)
    .style("font-family", "-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif")
    .style("overflow","visible");

  const g = svg.append("g").attr("transform", `translate(${marginL},${marginT})`);

  // Grid lines
  [1,2,3,4,5].forEach(v => {
    g.append("line")
      .attr("x1",0).attr("x2",plotW)
      .attr("y1",y(v)).attr("y2",y(v))
      .attr("stroke","#30363d").attr("stroke-dasharray","3,3");
    g.append("text")
      .attr("x",-6).attr("y",y(v)+4)
      .attr("text-anchor","end")
      .attr("font-size",10).attr("fill","#8b949e")
      .text(v);
  });

  // Bars
  cat_order.forEach(cat => {
    const catData = data.filter(d => d.category === cat);
    const gx = g.append("g").attr("transform", `translate(${x0(cat)},0)`);
    svcs.forEach(svc => {
      const d = catData.find(r => r.service === svc);
      if (!d) return;
      const bx = x1(svc);
      const by = y(d.mean_score);
      const bh = plotH - by;
      // Bar
      gx.append("rect")
        .attr("x",bx).attr("y",by)
        .attr("width",x1.bandwidth()).attr("height",bh)
        .attr("rx",3)
        .attr("fill", service_colors[svc])
        .attr("opacity", 0.85);
      // Value label
      if (bh > 14) {
        gx.append("text")
          .attr("x", bx + x1.bandwidth()/2).attr("y", by - 4)
          .attr("text-anchor","middle")
          .attr("font-size",9.5).attr("fill","#e6edf3")
          .text(d.mean_score.toFixed(2));
      }
    });
  });

  // X axis labels
  cat_order.forEach(cat => {
    g.append("text")
      .attr("transform", `translate(${x0(cat) + x0.bandwidth()/2},${plotH+12}) rotate(-30)`)
      .attr("text-anchor","end")
      .attr("font-size",11).attr("fill","#8b949e")
      .text(cat_labels_map[cat] || cat);
  });

  // Legend
  const lx = plotW - svcs.length * 90;
  svcs.forEach((svc, i) => {
    g.append("rect")
      .attr("x", lx + i*90).attr("y", plotH + 55)
      .attr("width",11).attr("height",11).attr("rx",2)
      .attr("fill", service_colors[svc]);
    g.append("text")
      .attr("x", lx + i*90 + 14).attr("y", plotH + 64)
      .attr("font-size",11).attr("fill","#8b949e")
      .text(service_labels[svc]);
  });

  return svg.node();
}

        .attr("x1",0).attr("x2",width)

    svg.append("rect")
      .attr("x", labelW + i*95).attr("y", height-22)
      .attr("width",11).attr("height",11).attr("rx",2)
      .attr("fill",service_colors[svc]).attr("opacity",0.85);
    svg.append("text")
      .attr("x", labelW + i*95 + 14).attr("y",height-12)
      .attr("font-size",11).attr("fill","#8b949e")
      .text(service_labels[svc]);
  });

  return svg.node();
}
Key finding: Models in the same family score identically (or near-identically) regardless of whether they run on GRIT or CIT — confirming that model quality is infrastructure-independent. The meaningful differences are in throughput and latency, which reflect the underlying hardware capacity of each service rather than model capability. The GPT-OSS pair is an exception: the 120B parameter version on CIT outperforms the 20B version on GRIT, but that reflects parameter count, not service quality.
