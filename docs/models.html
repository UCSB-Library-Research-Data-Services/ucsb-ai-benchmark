{
  if (!current_lb)
    return htl.html``;

  // Two-column layout: radar left, bar chart right
  const svc = selected_service;
  const col = service_colors[svc];

  // ─── Radar chart ────────────────────────────────────────────────────────
  const size   = 280;
  const cx     = size / 2;
  const cy     = size / 2;
  const r_max  = 100;
  const cats   = radar_data.filter(d => d.score !== null);
  const n      = cats.length;

  function angle(i) { return (i / n) * 2 * Math.PI - Math.PI / 2; }
  function pt(score, i, scale=1) {
    const a = angle(i);
    const r = r_max * (score / 5) * scale;
    return [cx + r * Math.cos(a), cy + r * Math.sin(a)];
  }

  const svgR = d3.create("svg")
    .attr("width", size).attr("height", size)
    .style("font-family","-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif");

  // Background rings
  [1,2,3,4,5].forEach(v => {
    const pts = cats.map((_,i) => pt(v,i)).map(p=>p.join(",")).join(" ");
    svgR.append("polygon").attr("points",pts)
      .attr("fill","none").attr("stroke","#30363d").attr("stroke-width",v===5?1.5:0.8);
    if (v % 2 === 0) {
      const [lx,ly] = pt(v, 0);
      svgR.append("text").attr("x",lx+3).attr("y",ly-2)
        .attr("font-size",8).attr("fill","#6e7681").text(v);
    }
  });

  // Spokes
  cats.forEach((_,i) => {
    const [x2,y2] = pt(5,i);
    svgR.append("line").attr("x1",cx).attr("y1",cy).attr("x2",x2).attr("y2",y2)
      .attr("stroke","#30363d").attr("stroke-width",0.8);
  });

  // Data polygon
  const polyPts = cats.map((d,i) => pt(d.score,i)).map(p=>p.join(",")).join(" ");
  svgR.append("polygon").attr("points",polyPts)
    .attr("fill",col).attr("fill-opacity",0.15)
    .attr("stroke",col).attr("stroke-width",2);

  // Data points
  cats.forEach((d,i) => {
    const [px,py] = pt(d.score,i);
    svgR.append("circle").attr("cx",px).attr("cy",py).attr("r",4)
      .attr("fill",col).attr("stroke","#0d1117").attr("stroke-width",1.5);
  });

  // Category labels
  cats.forEach((d,i) => {
    const [lx,ly] = pt(5.6, i);
    const lines = d.label.split(" ");
    lines.forEach((line, li) => {
      svgR.append("text")
        .attr("x",lx).attr("y",ly + li*12)
        .attr("text-anchor","middle").attr("dominant-baseline","middle")
        .attr("font-size",9).attr("fill","#8b949e")
        .text(line);
    });
  });

  // ─── Bar chart (right panel) ─────────────────────────────────────────────
  const bW   = 340;
  const bH   = size;
  const bPadL = 120, bPadR = 60, bPadT = 10, bPadB = 10;
  const barMaxW = bW - bPadL - bPadR;
  const rowH  = (bH - bPadT - bPadB) / cats.length;

  const svgB = d3.create("svg")
    .attr("width", bW).attr("height", bH)
    .style("font-family","-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif");

  const xS = d3.scaleLinear().domain([0,5]).range([0, barMaxW]);

  cats.forEach((d,i) => {
    const by = bPadT + i * rowH + rowH * 0.15;
    const bh = rowH * 0.7;
    const bx = bPadL;
    const scoreW = xS(d.score);

    // Background track
    svgB.append("rect").attr("x",bx).attr("y",by)
      .attr("width",barMaxW).attr("height",bh)
      .attr("rx",3).attr("fill","#1c2128");

Task-by-Task Results

Every individual evaluation run for this model. Scores are on a 1–5 scale. For literature synthesis, use the judge toggle to compare Gemini vs Claude evaluations. 

viewof judge_filter = Inputs.radio(
  ["claude-v4.6-sonnet", "gemini-3.5-flash", "all"],
  {
    label: "Judge (Lit. Synthesis)",
    value: "claude-v4.6-sonnet",
    format: v => v === "all" ? "Show both" : v
  }
)
{
  // Filter per-test rows with judge selection
  const rows = per_test.filter(d =>
    d.service === selected_service &&
    d.model   === selected_model
  ).filter(d => {
    if (d.category !== "literature_synthesis") return true;
    if (judge_filter === "all") return true;
    return (d.judge_model || "programmatic") === judge_filter ||
           (d.judge_model || "programmatic") === "programmatic";
  }).sort((a,b) => {
    const ci = cat_order.indexOf(a.category) - cat_order.indexOf(b.category);
    if (ci !== 0) return ci;
    return a.test_id.localeCompare(b.test_id);
  });

  if (rows.length === 0)
    return htl.html`<p style="color:#8b949e;padding:1rem">No tasks found for this selection.</p>`;

  function scoreHtml(s) {
    const cls = s >= 5 ? "score-5" : s >= 4 ? "score-4" : s >= 3 ? "score-3" : s >= 2 ? "score-2" : "score-1";
    return `<span class="${cls}" style="font-size:1.05rem;font-weight:700">${s}</span>`;
  }

  function judgeTag(j) {
    if (!j || j === "programmatic")
      return `<span style="font-size:0.72rem;color:#6e7681;background:rgba(255,255,255,0.05);padding:.1rem .4rem;border-radius:4px">programmatic</span>`;
    if (j.includes("claude"))
      return `<span style="font-size:0.72rem;color:#bc8cff;background:rgba(188,140,255,0.1);padding:.1rem .4rem;border-radius:4px">Claude</span>`;
    return `<span style="font-size:0.72rem;color:#58a6ff;background:rgba(88,166,255,0.1);padding:.1rem .4rem;border-radius:4px">Gemini</span>`;
  }

  function latencyFmt(l) {
    return l < 1 ? `${(l*1000).toFixed(0)}ms` : `${l.toFixed(1)}s`;
  }

  // Group by category
  const byCat = new Map();
  for (const r of rows) {
    if (!byCat.has(r.category)) byCat.set(r.category, []);
    byCat.get(r.category).push(r);
  }

  let html = `<div style="margin-top:0.5rem">`;

  for (const [cat, catRows] of byCat) {
    const svcCol = service_colors[selected_service];
    html += `
      <div style="margin-bottom:1.5rem">
        <div style="font-size:0.78rem;font-weight:700;text-transform:uppercase;
                    letter-spacing:.08em;color:${svcCol};
                    border-bottom:1px solid #30363d;padding-bottom:.4rem;
                    margin-bottom:.5rem">
          ${cat_labels_map[cat] || cat}
        </div>`;

    catRows.forEach(r => {
      // Truncate reason to ~180 chars
      const reason = r.reason ? r.reason.substring(0,200) + (r.reason.length > 200 ? "…" : "") : "";

      html += `
        <div style="display:grid;grid-template-columns:160px 48px 120px 90px 1fr;
                    gap:.5rem;align-items:start;
                    padding:.55rem .75rem;
                    border-radius:6px;
                    margin-bottom:3px;
                    background:rgba(255,255,255,0.02);
                    border:1px solid #30363d;
                    font-size:0.85rem">
          <div style="color:#e6edf3;font-weight:500">${r.test_id}</div>
          <div style="text-align:center">${scoreHtml(r.score)}</div>
          <div>${judgeTag(r.judge_model)}</div>
          <div style="color:#6e7681;font-size:0.8rem">
            ${latencyFmt(r.latency_sec)} &nbsp; ${r.tps.toFixed(0)} t/s
          </div>
          <div style="color:#8b949e;font-size:0.8rem;line-height:1.4">${reason}</div>
        </div>`;
    });

    html += `</div>`;
  }

  html += `</div>`;
  return htl.html([html]);
}

Judge Comparison — Literature Synthesis

For literature synthesis tasks, two LLM judges evaluated the same model outputs independently: Gemini 3.5-flash (development judge) and Claude v4.6-sonnet (production judge). Dots above the diagonal mean Gemini scored higher; below means Claude scored higher.

Judge Agreement — Gemini vs Claude Scores

Each dot = one literature synthesis task for the selected model. Diagonal = perfect agreement.
{
  const jRows = judge_cmp.filter(d =>

    const cy = pW - scale(cScore);
    const ptCol = agree ? "#3fb950" : "#f0883e";

    g.append("circle").attr("cx",cx).attr("cy",cy).attr("r",7)

     })
     .on("mouseout", function() {
       d3.select(this).attr("r",7).attr("opacity",0.85);
       tip.style("display","none");
     });
  });

  // Agreement legend
  const lx = pW + 12;
  [[  "#3fb950","Judges agree"],[  "#f0883e","Differ"]].forEach(([c,l],i) => {

     .attr("font-size",10).attr("fill","#8b949e").text(l);
  });

  const nAgree = jRows.filter(d => d.score_gemini === d.score_claude).length;

   .attr("text-anchor","middle").attr("font-size",10.5).attr("fill","#8b949e")
   .text(`${nAgree} / ${jRows.length} tasks in agreement (${Math.round(nAgree/jRows.length*100)}%)`);

  return svg.node();
