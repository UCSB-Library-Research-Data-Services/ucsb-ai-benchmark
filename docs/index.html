Infrastructure

Three UCSB-managed services were benchmarked, spanning two infrastructure models: on-premise Ollama clusters and cloud-hosted Amazon Bedrock.

GRIT

🖥️ On-premise · Ollama
llm.grit.ucsb.edu
UCSB's General Research IT cluster. Models are served locally via Ollama on UCSB-managed hardware. Largest model catalog — 14 models evaluated including vision-capable models.
14 models evaluated  ·  Budget: 10M tokens/month

CIT

🖥️ On-premise · Ollama
api.ai.college.ucsb.edu
College of Information Technology servers. Also Ollama-based, but shows notably higher throughput than GRIT for the same models — reflecting different hardware capacity. 6 models evaluated.
6 models evaluated  ·  Budget: 10M tokens/month

AICommons

☁️ Cloud · Amazon Bedrock
zkh52rh785.execute-api.us-east-1.amazonaws.com
AWS-hosted API with a diverse catalog including Claude, GPT-4o, Amazon Nova, and Llama 4. Currently 1 model evaluated (gpt-4o). Additional evaluations planned.
1 model evaluated  ·  Budget: 900K tokens/month

model_service_map = {
  const map = {};
  for (const row of lb) {
    if (!map[row.model]) map[row.model] = { model: row.model, services: [], infra: row.infra };
    map[row.model].services.push(row.service);
  }
  return map;
}

    // Cells

      cats_all.forEach((cat, ci) => {
        const cx = gx + ci * (cellSize + cellGap);
        let fill = "#1c2128";   // empty

        if (modelId) {

            opacity = 0.85;

            // model exists on service but wasn't tested on this category

            stroke  = "#30363d";
          }
          // else: model not on this service → stays empty
        }

        svg.append("rect")
          .attr("x", cx).attr("y", ry)
          .attr("width", cellSize).attr("height", cellSize)
          .attr("rx", 3)
          .attr("fill", fill)
          .attr("opacity", opacity)
          .attr("stroke", stroke)
          .attr("stroke-width", 0.5);
      });
    });
  });

  // ── Legend ───────────────────────────────────────────────────────────────

function scoreCell(v) {
  if (v === null || v === undefined) return `<span style="color:#6e7681">—</span>`;
  const cls = v >= 5 ? "score-5" : v >= 4 ? "score-4" : v >= 3 ? "score-3" : v >= 2 ? "score-2" : "score-1";
  return `<span class="${cls}">${v.toFixed(2)}</span>`;
}

function svcBadge(svc) {
  const cls = svc === "grit" ? "badge-grit" : svc === "CIT" ? "badge-cit" : "badge-aicommons";
  const lbl = svc === "grit" ? "GRIT" : svc === "CIT" ? "CIT" : "AICommons";
  return `<span class="service-badge ${cls}">${lbl}</span>`;
}

function infraTag(row) {
  if (row.infra_type === "onprem")
    return `<span style="font-size:0.72rem;color:#58a6ff">🖥 On-prem</span>`;
  return `<span style="font-size:0.72rem;color:#bc8cff">☁ Bedrock</span>`;
}
viewof sortCol = Inputs.select(
  ["overall_score", "avg_tps", "avg_latency_sec",
   "score_code_generation","score_metadata_generation",
   "score_literature_synthesis","score_data_interpretation",
   "score_ocr","score_vision_qa"],
  { value: "overall_score",

    format: v => ({
      overall_score: "Overall Score",

      avg_latency_sec: "Latency (sec)",
      score_code_generation: "Code Generation",
      score_metadata_generation: "Metadata Generation",
      score_literature_synthesis: "Literature Synthesis",
      score_data_interpretation: "Data Interpretation",
      score_ocr: "OCR",
      score_vision_qa: "Vision QA"
    })[v]
  }
)
{
  const sorted = [...lb].sort((a, b) => {
    const av = a[sortCol] ?? -1;
    const bv = b[sortCol] ?? -1;
    if (sortCol === "avg_latency_sec") return av - bv; // lower is better
    return bv - av;
  });

  const rows = sorted.map((r, i) => `
    <tr>
      <td style="color:#8b949e;font-size:0.8rem">${i + 1}</td>
      <td>${svcBadge(r.service)}</td>
      <td style="font-size:0.85rem;color:#e6edf3;font-weight:500">${r.model}</td>
      <td>${infraTag(r)}</td>
      <td style="text-align:center"><strong style="color:#FEBC11">${r.overall_score.toFixed(2)}</strong></td>
      ${cat_cols.map(c => `<td style="text-align:center">${scoreCell(r[c.key])}</td>`).join("")}
      <td style="text-align:right;color:#8b949e;font-size:0.85rem">${r.avg_tps.toFixed(0)}</td>
      <td style="text-align:right;color:#8b949e;font-size:0.85rem">${r.avg_latency_sec.toFixed(2)}s</td>
    </tr>
  `).join("");

  const html = `
    <table style="width:100%;border-collapse:collapse;">
      <thead>
        <tr>
          <th>#</th>
          <th>Service</th>
          <th>Model</th>
          <th>Infra</th>
          <th style="text-align:center">Overall</th>
          ${cat_cols.map(c => `<th style="text-align:center">${c.label}</th>`).join("")}
          <th style="text-align:right">TPS</th>
          <th style="text-align:right">Latency</th>
        </tr>
      </thead>
      <tbody>${rows}</tbody>
    </table>`;
  return html;
}
